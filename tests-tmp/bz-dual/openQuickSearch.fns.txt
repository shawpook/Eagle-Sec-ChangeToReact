  fns["openQuickSearch"] = function (...args) {
    try { initLinkVars(); } catch (err) { /* link var 初始化失败不阻塞（bundle 后备仍在） */ }
    const s = getScope();
    if (!s) return;
    return (function (event) {
            s.$root.$broadcast('OPEN_QUICK_SEARCH_MODAL');
        }).apply(null, args);
  };
