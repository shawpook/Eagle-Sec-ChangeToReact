  fns["zoom"] = function (...args) {
    try { initLinkVars(); } catch (err) { /* link var 初始化失败不阻塞（bundle 后备仍在） */ }
    const s = getScope();
    if (!s) return;
    return (function () {
            if (!s.isDetailMode) return;
            if (s.lastZoomMode === "edge") {
                if (s.current && !s.VIDEO_TYPES[s.current.ext]) {
                    s.zoomFitEdge();
                }
                else {
                    s.zoomFit();
                }
            }
            else {
                s.smartZoom();
            }
        }).apply(null, args);
  };
