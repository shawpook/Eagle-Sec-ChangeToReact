  fns["focusAppUnlockPassword"] = function (...args) {
    try { initLinkVars(); } catch (err) { /* link var 初始化失败不阻塞（bundle 后备仍在） */ }
    const s = getScope();
    if (!s) return;
    return (function () {
            setTimeout(() => {
                $("#app-lock-password-input").focus();
            }, 24);
            $("#app-lock-password-input").on("blur", () => {
                setTimeout(() => {
                    $("#app-lock-password-input").focus();
                }, 24);
            });
        }).apply(null, args);
  };
